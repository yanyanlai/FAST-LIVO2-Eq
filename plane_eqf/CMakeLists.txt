cmake_minimum_required(VERSION 3.10)
project(native_ray_plane_eqf LANGUAGES CXX)
if(NOT TARGET eqf_core)
  add_subdirectory(../eqf_core core)
endif()
add_library(plane_eqf ../src/eqf_plane/plane_eqf.cpp ../src/eqf_plane/prediction.cpp ../src/eqf_plane/update.cpp)
target_link_libraries(plane_eqf PUBLIC eqf_core)
target_include_directories(plane_eqf PUBLIC ../include)
target_compile_features(plane_eqf PUBLIC cxx_std_17)
target_compile_options(plane_eqf PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
option(PLANE_EQF_BUILD_TESTS "Build isolated native ray-plane prototype tests" ON)
if(PLANE_EQF_BUILD_TESTS)
  enable_testing()
  add_executable(plane_eqf_test ../tests/plane_eqf/test_plane_eqf.cpp)
  target_link_libraries(plane_eqf_test PRIVATE plane_eqf)
  add_test(NAME plane_eqf_contract COMMAND plane_eqf_test)
  add_executable(plane_eqf_oracle_test ../tests/plane_eqf/test_oracle.cpp)
  target_link_libraries(plane_eqf_oracle_test PRIVATE plane_eqf)
  target_compile_definitions(plane_eqf_oracle_test PRIVATE PLANE_EQF_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../tests/plane_eqf")
  add_test(NAME plane_eqf_oracle COMMAND plane_eqf_oracle_test)
  add_executable(plane_eqf_numerics_test ../tests/plane_eqf/test_numerics.cpp)
  target_link_libraries(plane_eqf_numerics_test PRIVATE plane_eqf)
  add_test(NAME plane_eqf_numerics COMMAND plane_eqf_numerics_test)
endif()
