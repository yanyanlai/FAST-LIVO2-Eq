cmake_minimum_required(VERSION 3.10)
project(fast_livo_eqf_ros LANGUAGES CXX)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs nav_msgs geometry_msgs cv_bridge)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(OpenCV 4 REQUIRED)
set(EQF_BUILD_TESTS ON CACHE BOOL "Build mathematical EqF tests")
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../eqf_core ${CMAKE_CURRENT_BINARY_DIR}/core)
add_library(eqf_frontend ${CMAKE_CURRENT_SOURCE_DIR}/../src/eqf_frontend.cpp)
target_include_directories(eqf_frontend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(eqf_frontend PUBLIC Eigen3::Eigen ${OpenCV_LIBS})
target_compile_features(eqf_frontend PUBLIC cxx_std_17)
add_library(eqf_ros_io ${CMAKE_CURRENT_SOURCE_DIR}/../src/eqf_ros_io.cpp)
target_include_directories(eqf_ros_io PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../include ${catkin_INCLUDE_DIRS})
target_link_libraries(eqf_ros_io PUBLIC Eigen3::Eigen ${catkin_LIBRARIES})
target_compile_features(eqf_ros_io PUBLIC cxx_std_17)
add_library(eqf_association_motion ${CMAKE_CURRENT_SOURCE_DIR}/../src/eqf_association_motion.cpp)
target_link_libraries(eqf_association_motion PUBLIC eqf_core)
add_executable(fastlivo_eqf_mapping ${CMAKE_CURRENT_SOURCE_DIR}/../src/eqf_mapping_node.cpp)
target_link_libraries(fastlivo_eqf_mapping PRIVATE eqf_core eqf_frontend eqf_ros_io eqf_association_motion)
add_executable(baseline_sensor_adapter ${CMAKE_CURRENT_SOURCE_DIR}/../src/baseline_sensor_adapter.cpp)
target_include_directories(baseline_sensor_adapter PRIVATE ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR}/../include)
target_link_libraries(baseline_sensor_adapter PRIVATE ${catkin_LIBRARIES} ${OpenCV_LIBS})
target_compile_features(baseline_sensor_adapter PRIVATE cxx_std_17)
if(CATKIN_DEVEL_PREFIX)
  set_target_properties(fastlivo_eqf_mapping PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CATKIN_DEVEL_PREFIX}/lib/fast_livo)
  set_target_properties(baseline_sensor_adapter PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CATKIN_DEVEL_PREFIX}/lib/fast_livo)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(eqf_core PRIVATE -fno-fast-math)
  target_compile_options(eqf_frontend PRIVATE -fno-fast-math)
  target_compile_options(eqf_ros_io PRIVATE -fno-fast-math)
  target_compile_options(eqf_association_motion PRIVATE -fno-fast-math)
  target_compile_options(fastlivo_eqf_mapping PRIVATE -fno-fast-math)
endif()
add_executable(eqf_ros_io_test ${CMAKE_CURRENT_SOURCE_DIR}/../tests/ros_io/test_ros_io.cpp)
target_link_libraries(eqf_ros_io_test PRIVATE eqf_ros_io)
enable_testing()
add_test(NAME eqf_ros_io COMMAND eqf_ros_io_test)
add_executable(baseline_adapter_test ${CMAKE_CURRENT_SOURCE_DIR}/../tests/ros_io/test_baseline_adapter.cpp)
target_include_directories(baseline_adapter_test PRIVATE ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR}/../include)
target_link_libraries(baseline_adapter_test PRIVATE ${catkin_LIBRARIES} ${OpenCV_LIBS})
target_compile_features(baseline_adapter_test PRIVATE cxx_std_17)
add_test(NAME baseline_adapter COMMAND baseline_adapter_test)
add_executable(eqf_fisheye_test ${CMAKE_CURRENT_SOURCE_DIR}/../tests/frontend/test_fisheye.cpp)
target_link_libraries(eqf_fisheye_test PRIVATE eqf_frontend)
add_test(NAME eqf_fisheye COMMAND eqf_fisheye_test)
add_executable(eqf_frontend_tests ${CMAKE_CURRENT_SOURCE_DIR}/../tests/frontend/test_frontend.cpp)
target_link_libraries(eqf_frontend_tests PRIVATE eqf_frontend)
add_test(NAME eqf_frontend COMMAND eqf_frontend_tests)
add_executable(eqf_lidar_surface_tests ${CMAKE_CURRENT_SOURCE_DIR}/../tests/frontend/test_lidar_surface.cpp)
target_link_libraries(eqf_lidar_surface_tests PRIVATE eqf_frontend)
add_test(NAME eqf_lidar_surface COMMAND eqf_lidar_surface_tests)
add_executable(eqf_association_motion_test ${CMAKE_CURRENT_SOURCE_DIR}/../tests/frontend/test_association_motion.cpp)
target_link_libraries(eqf_association_motion_test PRIVATE eqf_association_motion)
add_test(NAME eqf_association_motion COMMAND eqf_association_motion_test)

option(FASTLIVO_BUILD_PLANE_PROTOTYPE "Build experimental ray-plane observation-input LIO mode" OFF)
if(FASTLIVO_BUILD_PLANE_PROTOTYPE)
  find_package(Boost 1.65 REQUIRED)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../plane_eqf ${CMAKE_CURRENT_BINARY_DIR}/plane_core)
  add_library(eqf_plane_frontend ${CMAKE_CURRENT_SOURCE_DIR}/../src/eqf_plane_frontend.cpp)
  target_include_directories(eqf_plane_frontend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../include ${Boost_INCLUDE_DIRS})
  target_link_libraries(eqf_plane_frontend PUBLIC Eigen3::Eigen)
  target_compile_features(eqf_plane_frontend PUBLIC cxx_std_17)
  add_executable(fastlivo_eqf_plane_mapping ${CMAKE_CURRENT_SOURCE_DIR}/../src/eqf_plane_mapping_node.cpp)
  target_link_libraries(fastlivo_eqf_plane_mapping PRIVATE plane_eqf eqf_plane_frontend eqf_association_motion eqf_ros_io)
  add_executable(eqf_plane_frontend_test ${CMAKE_CURRENT_SOURCE_DIR}/../tests/plane_frontend/test_plane_frontend.cpp)
  target_link_libraries(eqf_plane_frontend_test PRIVATE eqf_plane_frontend)
  add_test(NAME eqf_plane_frontend COMMAND eqf_plane_frontend_test)
  if(CATKIN_DEVEL_PREFIX)
    set_target_properties(fastlivo_eqf_plane_mapping PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CATKIN_DEVEL_PREFIX}/lib/fast_livo)
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(plane_eqf PRIVATE -fno-fast-math)
    target_compile_options(eqf_plane_frontend PRIVATE -fno-fast-math)
    target_compile_options(fastlivo_eqf_plane_mapping PRIVATE -fno-fast-math)
  endif()
endif()
