cmake_minimum_required(VERSION 3.10)
project(eqf_joint_plane_birth LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(eqf_plane_birth ../src/eqf_plane_birth/joint_gaussian.cpp)
target_include_directories(eqf_plane_birth PUBLIC ../include)
target_link_libraries(eqf_plane_birth PUBLIC Eigen3::Eigen)
target_compile_features(eqf_plane_birth PUBLIC cxx_std_17)
target_compile_options(eqf_plane_birth PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
add_library(eqf_plane_birth_ray ../src/eqf_plane_birth/native_range.cpp)
target_include_directories(eqf_plane_birth_ray PUBLIC ../include)
target_link_libraries(eqf_plane_birth_ray PUBLIC eqf_plane_birth)
target_compile_features(eqf_plane_birth_ray PUBLIC cxx_std_17)
target_compile_options(eqf_plane_birth_ray PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
option(PLANE_BIRTH_BUILD_TESTS "Build isolated Gaussian plane-birth tests" ON)
if(PLANE_BIRTH_BUILD_TESTS)
  enable_testing()
  add_executable(plane_birth_test ../tests/plane_birth/test_joint_gaussian.cpp)
  target_link_libraries(plane_birth_test PRIVATE eqf_plane_birth)
  foreach(case information psd_clone coordinate gauge rejection exact_three static_independent)
    add_test(NAME plane_birth_${case} COMMAND plane_birth_test ${case})
  endforeach()
  add_executable(plane_birth_ray_test ../tests/plane_birth/test_native_range.cpp)
  target_link_libraries(plane_birth_ray_test PRIVATE eqf_plane_birth_ray)
  foreach(case derivative finite_action gauge rejection covariance rank)
    add_test(NAME plane_birth_ray_${case} COMMAND plane_birth_ray_test ${case})
  endforeach()
endif()
