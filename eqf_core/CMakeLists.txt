cmake_minimum_required(VERSION 3.10)
project(strict_landmark_eqf LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(eqf_core ../src/eqf/state.cpp ../src/eqf/lie.cpp ../src/eqf/covariance.cpp
  ../src/eqf/landmarks.cpp ../src/eqf/prediction.cpp ../src/eqf/correction.cpp ../src/eqf/measurement.cpp)
target_include_directories(eqf_core PUBLIC ../include)
target_link_libraries(eqf_core PUBLIC Eigen3::Eigen)
if(TARGET OpenMP::OpenMP_CXX AND FASTLIVO_UPSTREAM_EQF)
  target_link_libraries(eqf_core PUBLIC OpenMP::OpenMP_CXX)
endif()
target_compile_features(eqf_core PUBLIC cxx_std_17)
target_compile_options(eqf_core PRIVATE -Wall -Wextra -Wpedantic)
option(EQF_BUILD_TESTS "Build standalone EqF tests" ON)
if(EQF_BUILD_TESTS)
  add_executable(eqf_core_test ../tests/eqf/test_core.cpp)
  target_link_libraries(eqf_core_test PRIVATE eqf_core)
  enable_testing()
  add_test(NAME eqf_core COMMAND eqf_core_test)
  add_executable(eqf_filter_test ../tests/eqf/test_filter.cpp)
  target_link_libraries(eqf_filter_test PRIVATE eqf_core)
  add_test(NAME eqf_filter COMMAND eqf_filter_test)
  add_executable(eqf_oracle_test ../tests/eqf/test_oracle.cpp)
  target_link_libraries(eqf_oracle_test PRIVATE eqf_core)
  target_compile_definitions(eqf_oracle_test PRIVATE EQF_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../tests/eqf")
  add_test(NAME eqf_oracle COMMAND eqf_oracle_test)
  add_executable(eqf_math_test ../tests/eqf/test_math.cpp)
  target_link_libraries(eqf_math_test PRIVATE eqf_core)
  add_test(NAME eqf_math COMMAND eqf_math_test)
  add_executable(eqf_physical_kinematics_regression ../tests/performance/test_physical_pullback.cpp)
  target_link_libraries(eqf_physical_kinematics_regression PRIVATE eqf_core)
  add_test(NAME eqf_physical_chart COMMAND eqf_physical_kinematics_regression)
  add_executable(eqf_visual_section_regression ../tests/performance/test_visual_section.cpp)
  target_link_libraries(eqf_visual_section_regression PRIVATE eqf_core)
  add_test(NAME eqf_visual_section COMMAND eqf_visual_section_regression)
  add_executable(eqf_core_benchmark ../tests/eqf/benchmark_core.cpp)
  target_link_libraries(eqf_core_benchmark PRIVATE eqf_core)
endif()
